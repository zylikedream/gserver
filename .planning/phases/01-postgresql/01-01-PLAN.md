```go
package gxypgx

import (
    "context"
    "time"
    "github.com/jackc/pgx/v5/pgxpool"
    "github.com/gogf/gf/v2/frame/g"
    "github.com/gogf/gf/v2/os/gctx"
    "github.com/gogf/gf/v2/os/glog"
)

type pgxApp struct {
    gxyapp.App
    pool *pgxpool.Pool
    conf *pgxConfig
}

type pgxConfig struct {
    URL string `toml:"url"`
}

var pgxAppInstance *pgxApp

func PGX() *pgxApp {
    return pgxAppInstance
}

func NewPGXApp() *pgxApp {
    cfg := g.Cfg()
    conf := &pgxConfig{}
    ctx := gctx.New()
    if err := gxyutil.CfgUnmarshalKey(ctx, cfg, "postgres", conf); err != nil {
        glog.Fatal(ctx, err)
    }
    pgxAppInstance = &pgxApp{conf: conf}
    return pgxAppInstance
}

func (p *pgxApp) OnModInit(ctx context.Context) error {
    poolConfig, err := pgxpool.ParseConfig(p.conf.URL)
    if err != nil {
        return err
    }
    p.pool, err = pgxpool.NewWithConfig(ctx, poolConfig)
    if err != nil {
        return err
    }
    return nil
}

func (p *pgxApp) OnModStart(ctx context.Context) error {
    if err := p.pool.Ping(ctx); err != nil {
        glog.Fatal(ctx, err)
    }
    glog.Infof(ctx, "[module]postgres start success: %s", p.conf.URL)
    return nil
}

func (p *pgxApp) OnModStop(ctx context.Context) error {
    if p.pool != nil {
        p.pool.Close()
    }
    glog.Info(ctx, "[module]postgres stop success")
    return nil
}

func (p *pgxApp) GetPool() *pgxpool.Pool {
    return p.pool
}
```

**关键实现细节：**
1. 使用 `pgxpool.ParseConfig` 解析 Connection URL（D-01）
2. 连接池参数通过 URL 配置（如 `?pool_max_conns=20&pool_min_conns=5`）（D-02）
3. 配置节命名为 `[postgres]`（D-03）
4. 连接失败时调用 `glog.Fatal` 快速失败（D-09）
5. 实现 `OnModInit`, `OnModStart`, `OnModStop` 生命周期方法
6. 添加 `util` 包导入（需要 `gserver/util`）

```go
package gxypgx

import (
    "context"
    "github.com/jackc/pgx/v5/pgxpool"

// CreateTable 执行 CREATE TABLE 语句
func (p *pgxApp) CreateTable(ctx context.Context, sql string) error {
    _, err := p.pool.Exec(ctx, sql)
    return err
}

// CreateIndex 执行 CREATE INDEX 语句
func (p *pgxApp) CreateIndex(ctx context.Context, sql string) error {
    _, err := p.pool.Exec(ctx, sql)
    return err
}

// TableExists 检查表是否存在
func (p *pgxApp) TableExists(ctx context.Context, tableName string) (bool, error) {
    var exists bool
    query := `
        SELECT EXISTS (
            SELECT FROM information_schema.tables
            WHERE table_schema = 'public'
            AND table_name = $1
        )
    `
    err := p.pool.QueryRow(ctx, query, tableName).Scan(&exists)
    return exists, err
}

// IndexExists 检查索引是否存在
func (p *pgxApp) IndexExists(ctx context.Context, indexName string) (bool, error) {
    var exists bool
    query := `
        SELECT EXISTS (
            SELECT FROM pg_indexes
            WHERE schemaname = 'public'
            AND indexname = $1
        )
    `
    err := p.pool.QueryRow(ctx, query, indexName).Scan(&exists)
    return exists, err
}
```

**关键实现细节：**
1. 使用 `p.pool.Exec` 执行 DDL 语句

3. 使用参数化查询防止 SQL 注入
4. 错误直接返回 pgx 错误（D-10）

```go
package gxypgx

import (
    "context"
    "github.com/jackc/pgx/v5/pgxpool"
)

// InsertOne 插入单行数据
func (p *pgxApp) InsertOne(ctx context.Context, tableName string, columns []string, values []interface{}) error {
    query := p.buildInsertQuery(tableName, columns)
    _, err := p.pool.Exec(ctx, query, values...)
    return err
}

// FindOne 查询单行数据
func (p *pgxApp) FindOne(ctx context.Context, tableName string, dest interface{}, where string, args ...interface{}) error {
    query := "SELECT * FROM " + tableName + " WHERE " + where
    row := p.pool.QueryRow(ctx, query, args...)
    return row.Scan(dest)
}

// Update 更新数据
func (p *pgxApp) Update(ctx context.Context, tableName string, set string, where string, args ...interface{}) error {
    query := "UPDATE " + tableName + " SET " + set + " WHERE " + where
    _, err := p.pool.Exec(ctx, query, args...)
    return err
}

// Delete 删除数据
func (p *pgxApp) Delete(ctx context.Context, tableName string, where string, args ...interface{}) error {
    query := "DELETE FROM " + tableName + " WHERE " + where
    _, err := p.pool.Exec(ctx, query, args...)
    return err
}

// buildInsertQuery 构建 INSERT 语句（占位符格式 $1, $2, ...）
func (p *pgxApp) buildInsertQuery(tableName string, columns []string) string {
    query := "INSERT INTO " + tableName + " ("
    for i, col := range columns {
        if i > 0 {
            query += ", "
        }
        query += col
    }
    query += ") VALUES ("
    for i := 0; i < len(columns); i++ {
        if i > 0 {
            query += ", "
        }
        query += "$" + string(rune('1'+i))
    }
    query += ")"
    return query
}
```

**关键实现细节：**
1. 使用 pgx 的参数化查询（$1, $2, ...）而非 ?
2. 错误直接返回 pgx 错误（D-10）
3. 方法签名参考 MongoDB 实现（保持一致性）
4. Phase 1 不支持事务（D-08）

**注意：** 这是基础 CRUD 方法，后续 Phase 2 会根据实际需求优化（如支持结构体映射）。
